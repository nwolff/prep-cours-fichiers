# en commentaire, l'instruction qu'ils codent, écrite en assembleur.

20 00 80 d2  # mov x0, #0x1

c2 01 80 d2  # mov x2, #0xe
01 10 00 d4  # svc #0x80
30 00 80 d2  # mov x16, #0x1
00 00 80 d2  # mov x0, #0x0
01 10 00 d4  # svc #0x80
